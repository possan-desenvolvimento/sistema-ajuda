💡 O que é SQL?
SQL (Structured Query Language) é a linguagem padrão para gerenciar e manipular bancos de dados relacionais. Com ela você pode SELECT (consultar), INSERT (inserir), UPDATE (atualizar) e DELETE (remover) dados.
📊 SELECT Consultar dados ▶
SELECT * FROM usuarios; Seleciona todas as colunas e linhas da tabela
SELECT nome, email FROM usuarios; Seleciona apenas colunas específicas
SELECT * FROM usuarios WHERE idade > 18; Filtra registros com condição (maior que 18)
SELECT * FROM usuarios ORDER BY nome ASC; Ordena resultados por nome (A-Z)
SELECT * FROM usuarios LIMIT 10; Limita a quantidade de resultados (10 primeiros)
SELECT COUNT(*) FROM usuarios; Conta o total de registros na tabela
SELECT DISTINCT cidade FROM usuarios; Retorna apenas valores únicos (sem duplicados)
SELECT * FROM usuarios WHERE nome LIKE 'A%'; Busca nomes que começam com 'A'
SELECT * FROM usuarios WHERE idade BETWEEN 18 AND 30; Filtra idades entre 18 e 30 (inclusive)
SELECT * FROM usuarios WHERE id IN (1, 5, 10); Filtra por múltiplos IDs específicos
SELECT * FROM usuarios WHERE email IS NULL; Busca registros onde email está vazio (NULL)
SELECT AVG(idade) FROM usuarios; Calcula a média de idade
➕ INSERT Inserir dados ▶
INSERT INTO usuarios (nome, email) VALUES ('João', 'joao@email.com'); Insere um novo registro com nome e email
INSERT INTO usuarios (nome, idade) VALUES ('Maria', 25); Insere apenas alguns campos (os outros ficam NULL)
INSERT INTO usuarios VALUES (1, 'Pedro', 'pedro@email.com', 30); Insere com todas as colunas na ordem da tabela
INSERT INTO usuarios (nome, email) VALUES ('Ana', 'ana@email.com'), ('Carlos', 'carlos@email.com'); Insere múltiplos registros de uma vez
INSERT INTO usuarios (nome, email, idade) VALUES ('Julia', 'julia@email.com', 22); Insere com todos os campos preenchidos
INSERT INTO usuarios_backup SELECT * FROM usuarios WHERE idade > 30; Copia registros filtrados para outra tabela
INSERT INTO usuarios (nome, email) SELECT nome, email FROM usuarios_antigos WHERE ativo = 1; Insere dados de uma tabela para outra com filtro
✏️ UPDATE Atualizar dados ▶
UPDATE usuarios SET email = 'novo@email.com' WHERE id = 1; Atualiza o email de um usuário específico
UPDATE usuarios SET idade = 26 WHERE nome = 'Maria'; Atualiza idade de um usuário pelo nome
UPDATE usuarios SET status = 'ativo', updated_at = NOW() WHERE id > 10; Atualiza múltiplos campos de uma vez
UPDATE usuarios SET idade = idade + 1 WHERE YEAR(data_nasc) < 2000; Incrementa idade baseado na data de nascimento
UPDATE usuarios SET email = CONCAT(nome, '@empresa.com'); Gera email automaticamente a partir do nome
🗑️ DELETE Remover dados ▶
DELETE FROM usuarios WHERE id = 1; Remove um usuário específico pelo ID
DELETE FROM usuarios WHERE idade < 18; Remove todos os usuários menores de 18 anos
DELETE FROM usuarios WHERE status = 'inativo'; Remove todos os usuários inativos
DELETE FROM usuarios WHERE data_criacao < '2024-01-01'; Remove registros antigos (antes de 2024)
🏗️ CREATE TABLE Criar tabela ▶
CREATE TABLE usuarios (id INT PRIMARY KEY, nome VARCHAR(100), email VARCHAR(100)); Tabela básica com chave primária
CREATE TABLE produtos (id SERIAL, nome TEXT, preco DECIMAL(10,2), estoque INT); Tabela com auto-increment (SERIAL no PostgreSQL)
CREATE TABLE pedidos (id INT, usuario_id INT, data DATE, FOREIGN KEY (usuario_id) REFERENCES usuarios(id)); Tabela com chave estrangeira (relacionamento)
CREATE TABLE categorias (id INT PRIMARY KEY, nome VARCHAR(50), descricao TEXT); Tabela simples para categorias
CREATE TABLE logs (id SERIAL, usuario_id INT, acao TEXT, data TIMESTAMP DEFAULT NOW()); Tabela com valor padrão (NOW() = data atual)
🔧 ALTER TABLE Modificar estrutura ▶
ALTER TABLE usuarios ADD COLUMN telefone VARCHAR(20); Adiciona uma nova coluna à tabela
ALTER TABLE usuarios DROP COLUMN telefone; Remove uma coluna da tabela
ALTER TABLE usuarios RENAME COLUMN nome TO nome_completo; Renomeia uma coluna
ALTER TABLE usuarios ALTER COLUMN idade SET NOT NULL; Torna a coluna obrigatória (não aceita NULL)
ALTER TABLE usuarios ADD CONSTRAINT unique_email UNIQUE (email); Adiciona restrição UNIQUE (não permite duplicados)
💥 DROP Remover tabelas ▶
DROP TABLE usuarios; Remove a tabela permanentemente
DROP TABLE IF EXISTS usuarios; Remove a tabela se ela existir (evita erro)
DROP DATABASE meu_banco; Remove o banco de dados inteiro
DROP TABLE pedidos, produtos, categorias; Remove múltiplas tabelas de uma vez
🔗 JOINS Relacionar tabelas ▶
SELECT * FROM usuarios INNER JOIN pedidos ON usuarios.id = pedidos.usuario_id; Retorna apenas registros com correspondência em ambas tabelas
SELECT * FROM usuarios LEFT JOIN pedidos ON usuarios.id = pedidos.usuario_id; Retorna todos os usuários, mesmo sem pedidos (NULL onde não tem)
SELECT u.nome, p.total FROM usuarios u JOIN pedidos p ON u.id = p.usuario_id WHERE p.total > 100; JOIN com alias (u, p) e filtro
SELECT * FROM usuarios u LEFT JOIN pedidos p ON u.id = p.usuario_id LEFT JOIN produtos pr ON p.produto_id = pr.id; Múltiplos JOINs (usuários, pedidos, produtos)
📈 GROUP BY & HAVING Agrupar dados ▶
SELECT cidade, COUNT(*) FROM usuarios GROUP BY cidade; Conta quantos usuários por cidade
SELECT cidade, COUNT(*) FROM usuarios GROUP BY cidade HAVING COUNT(*) > 5; Filtra cidades com mais de 5 usuários
SELECT YEAR(data_criacao) AS ano, COUNT(*) FROM usuarios GROUP BY YEAR(data_criacao); Conta usuários criados por ano
SELECT status, AVG(idade) FROM usuarios GROUP BY status; Média de idade por status
